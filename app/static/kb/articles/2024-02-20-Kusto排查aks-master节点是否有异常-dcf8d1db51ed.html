//Black box monitoring for cluster

cluster("akscn").database("AKSprod").BlackboxMonitoringActivity

| where PreciseTimeStamp > ago(2d)

| where fqdn contains "smzdm-pre-dns-5198f1dc.hcp.chinanorth2.cx.prod.service.azk8s.cn" 集群FQDN

| where state != "Healthy"

| project PreciseTimeStamp, state, provisioningState, reason, agentNodeCount, msg, resourceGroupName, resourceName, underlayName

| order by PreciseTimeStamp asc

| distinct reason

| render timeline

//Black box monitoring using fqdn to find where cluster is not healthy

| where fqdn == "smzdm-pre-dns-5198f1dc.hcp.chinanorth2.cx.prod.service.azk8s.cn" 集群FQDN

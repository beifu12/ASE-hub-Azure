脚本参考如下：

1，在自动化账户创建powershell类型的runbook，并粘贴如下脚本，修改对应您环境中的值。

如下标黄需要替换为您环境中的值。

$Cred=Get-AutomationPSCredential -Name "zhangnan";              ##凭据的名称

Connect-AzAccount -Credential $Cred -EnvironmentName AzureChinaCloud

Select-AzSubscription -SubscriptionId "57e04838-xxxx-41d73c5154c0"     ##订阅id

$user=Get-AzADGroupMember -GroupObjectId "f7e5aa79-xxxx-ac38-ed00a8611826"  ##AAD Group的object id

$username=$user.DisplayName

$exportCSV=New-Object -TypeName System.Collections.ArrayList

$exportCSV.Add($username)

将获取的有效信息通过邮件通知管理员

$recipient="zhang.nan6@oe.21vianet.com"  ##收件箱

$emailFrom="sisiyxxx012@163.com"    ##发件箱

$smtpServer="smtp.163.com"   ##发件邮箱服务器

$smtpPort="25"      ##端口号

$smtpCredential=new-object -typename System.Management.Automation.PSCredential -argumentlist "sisixxxx012@163.com",(ConvertTo-SecureString "AHAVFxxxxYGED" -AsPlainText -Force)    ##发件箱地址以及授权码，我这里使用的163邮箱，授权码需要进行额外设置。

$emailBody='Please check the Group User'+[Environment]::NewLine+$exportCSV[0];

Send-MailMessage -Subject 'Please check the Group User' -Body $emailBody -To $recipient -From $emailFrom -SmtpServer $smtpServer -Credential $smtpCredential -Port $smtpPort -UseSsl

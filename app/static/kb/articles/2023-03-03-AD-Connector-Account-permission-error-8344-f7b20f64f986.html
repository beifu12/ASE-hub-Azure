Set-ADSyncBasicReadPermissions -ADConnectorAccountDN "CN=leoadsync,CN=Users,DC=leo2022,DC=com"

Set-ADSyncPasswordHashSyncPermissions -ADConnectorAccountDN "CN=leoadsync,CN=Users,DC=leo2022,DC=com"

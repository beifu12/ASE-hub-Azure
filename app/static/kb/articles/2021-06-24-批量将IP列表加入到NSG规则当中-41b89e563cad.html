为多个虚拟机的NSG添加此规则

#首先，需要贵方进行编辑附件的CSV文件，将后期需要添加的虚拟机所使用的NSG 的名称、资源组名称完善在此CSV当中。

CSV参阅附件。

#导入CSV文件，获取配置：

Import-Csv E:\123\NSGs.csv

#获取CDN 源站IP范围，赋值给变量$iprange_prefix：

$iprange_prefix="121.43.18.0/24,120.25.115.0/24,101.200.106.0/24,120.55.177.0/24,120.27.173.0/24,120.55.107.0/24,123.57.117.0/24,120.76.16.0/24,182.92.253.32/27,60.205.193.64/27,60.205.193.96/27,120.78.44.128/26,118.178.15.0/24,39.106.237.192/26,106.15.101.96/27,47.101.16.64/27,47.106.31.0/24,47.98.74.0/25,47.97.242.96/27,112.124.159.0/24,39.96.130.0/24,39.96.119.0/24,47.99.20.0/24,47.104.53.0/26,47.108.23.192/26,39.104.199.128/26,39.96.158.0/24,47.110.182.0/24,120.77.139.0/25,47.102.187.0/25"

#转换为数组：

$iprange_prefix2 =  $iprange_prefix -split ","

#添加入站规则配置，端口号请修改确认：

Add-AzureRmNetworkSecurityRuleConfig -Name CDN_WAF_Rules -NetworkSecurityGroup $nsg -Description "This_is_CDN_Source_IP_Range" -Protocol Tcp -SourcePortRange * -DestinationPortRange 8086 –SourceAddressPrefix $iprange_prefix2  -DestinationAddressPrefix * -Access Allow -Priority 2000 -Direction Inbound

#使配置生效：

#循环的脚本如下：

Import-Csv -Path E:\123\NSGs.csv | % { $nsg = Get-AzureRmNetworkSecurityGroup -ResourceGroupName $_.RESOURCEGROUP -Name $_.NAME;

$iprange_prefix2 =  $iprange_prefix -split ",";

Add-AzureRmNetworkSecurityRuleConfig -Name CDN_WAF_Rules2 -NetworkSecurityGroup $nsg -Description "This_is_CDN_Source_IP_Range" -Protocol Tcp -SourcePortRange * -DestinationPortRange 8086 -SourceAddressPrefix $iprange_prefix2 -DestinationAddressPrefix * -Access Allow -Priority 2002 -Direction Inbound ;

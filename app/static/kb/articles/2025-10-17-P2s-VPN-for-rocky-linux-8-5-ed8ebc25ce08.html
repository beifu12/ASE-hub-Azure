#Install strongswan and dependency packages

dnf install epel-release

dnf install strongswan

dnf install tpm2*

generate CA RSA Key pairs

strongswan pki --gen rsa --outform pem > caKey.pem

strongswan pki --self --in caKey.pem --dn "CN=VPN CA" --ca --outform pem > caCert.pem

export cer file of CA and upload to Azure VPN gateway

openssl x509 -in caCert.pem -outform der | base64 -w0 ; echo

genarate vpn client key pairs

export PASSWORD="password" # replace with the real password

export USERNAME=$(hostnamectl --static)

strongswan pki --gen --outform pem > "${USERNAME}Key.pem"

strongswan pki --pub --in "${USERNAME}Key.pem" | strongswan pki --issue --cacert caCert.pem --cakey caKey.pem --dn "CN=${USERNAME}" --san "${USERNAME}" --flag clientAuth --outform pem > "${USERNAME}Cert.pem"

openssl pkcs12 -in "${USERNAME}Cert.pem" -inkey "${USERNAME}Key.pem" -certfile caCert.pem -export -out "${USERNAME}.p12" -password "pass:${PASSWORD}"

cp $(USERNAME).p12 /etc/strongswan/ipsec.d/private/

#download VPN client configuration file, after decompression, upload VpnServerRoot.cer_0 to the local device

cp VpnServerRoot.cer_0 /etc/strongswan/ipsec.d/cacerts/

edit the ipsec configuration file and add the new IPSEC connction

vi /etc/strongswan/ipsec.conf

conn azure

keyexchange=ikev2

type=tunnel

leftfirewall=yes

left=%any

leftcert=$(USRENAME)Cert.pem #replase by value of variable USRENAME

leftauth=eap-tls

leftid=%$(USERNAME) #replase by value of variable USRENAME

right=azuregateway-3dfe0023-a199-4d19-a19a-331e4cf5449a-87b9da247488.vpn.chinacloudapi.cn #replcace by VPN gateway FQDN

rightid=%azuregateway-3dfe0023-a199-4d19-a19a-331e4cf5449a-87b9da247488.vpn.chinacloudapi.cn #replcace by VPN gateway FQDN

rightsubnet=10.0.0.0/16,172.25.0.0/16 #replcas by Azure network address range

leftsourceip=%config

auto=start

auto=route

keyingtries=%forever

esp=aes256gcm16

add the auth file,type the P12 file name and password

vi /etc/strongswan/ipsec.secrets

: P12 $(USERNAME).p12 'password' # replase by value of variable USRENAME

#initial connection

strongswan restart

strongswan up azure

经过验证可以使用下面的AzureAD PS脚本来批量修改Entra ID用户的密码,：

连接到 Azure AD

Connect-AzureAD -AzureEnvironmentName azurechinacloud

设置目标用户列表

$userList = @(

@{ UserPrincipalName = "pxxx1@paulww1.top"; NewPassword = "H/3xxxxxxxxxxxod2" },

@{ UserPrincipalName = "pxxx2@paulww1.top"; NewPassword = "H/3xxxxxxxxxxxod2" }

)

遍历用户列表并重置密码

foreach ($user in $userList) {

try {

获取用户对象

$azureUser = Get-AzureADUser -ObjectId $user.UserPrincipalName

if ($azureUser) {

将明文密码转换为 SecureString

$securePassword = ConvertTo-SecureString -String $user.NewPassword -AsPlainText -Force

重设新密码，且不要求用户在首次登录时更改密码

Set-AzureADUserPassword -ObjectId $azureUser.ObjectId -Password $securePassword -ForceChangePasswordNextLogin $false

Write-Host "Password reset successfully for $($user.UserPrincipalName)"

} else {

Write-Host "User $($user.UserPrincipalName) not found."

}

} catch {

Write-Host "Error resetting password for $($user.UserPrincipalName): $_"

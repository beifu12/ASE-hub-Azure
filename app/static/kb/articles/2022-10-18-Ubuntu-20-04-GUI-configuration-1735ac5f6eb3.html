#安装ubuntu-desktop

apt update

apt install ubuntu-desktop

#安装xubuntu-desktop

apt install xubuntu-desktop

#继续安装 xrdp

apt install xrdp

systemctl status xrdp

#添加用户xrdp和虚拟机创建时输入的用户

adduser xrdp ssl-cert

adduser paul ssl-cert

#重启xrdp

systemctl restart xrdp

#参考下面网站链接创建两个文件并添加对应内容

https://blog.csdn.net/qq_37556330/article/details/116168449

vim /etc/polkit-1/localauthority/50-local.d/45-allow-colord.pkla

[Allow Colord all Users]

Identity=unix-user:*

Action=org.freedesktop.color-manager.create-device;org.freedesktop.color-manager.create-profile;org.freedesktop.color-manager.delete-device;org.freedesktop.color-manager.delete-profile;org.freedesktop.color-manager.modify-device;org.freedesktop.color-manager.modify-profile

ResultAny=no

ResultInactive=no

ResultActive=yes

vim /etc/polkit-1/localauthority/50-local.d/46-allow-packagekit.pkla

[Allow Refresh Repository all Users]

Action=org.freedesktop.packagekit.system-sources-refresh

#重启机器

reboot

class AzureCDNAuthorization:

"""Class to handle Azure CDN API authorization using HMAC SHA256."""

def __init__(self, key_id, key_value):

"""Initialize with key ID and key value."""

self.key_id = key_id

self.key_value = key_value

def calculate_header(self, request_url, request_time, http_method):

"""Calculate and return the Authorization header for Azure CDN API requests."""

Parse URL and sort query parameters

urlparts = urllib.parse.urlparse(request_url)

queries = urllib.parse.parse_qs(urlparts.query)

ordered_queries = OrderedDict(sorted(queries.items()))

Construct message to hash

message = "%s\r\n%s\r\n%s\r\n%s" % (

urlparts.path,

", ".join(['%s:%s' % (key, value[0]) for (key, value) in ordered_queries.items()]),

request_time,

http_method

)

Calculate digest and format header

digest = hmac.new(

bytearray(self.key_value, "utf-8"),

bytearray(message, "utf-8"),

hashlib.sha256

).hexdigest().upper()

return "AzureCDN %s:%s" % (self.key_id, digest)

def cdnTraffic():

"""Check CDN traffic and generate alerts if thresholds are exceeded."""

now = datetime.now(timezone.utc)

Define start and end times for traffic data retrieval

startTime = now - timedelta(minutes=60)

endTime = now - timedelta(minutes=54)

Iterate over CDN endpoints

Fetch traffic data and analyze differences

return alert_body

经测试验证可以通过如下Azure Powershell命令创建符合贵司要求的虚拟机（指定已有资源组、虚拟网络、网络安全组、虚拟机大小、磁盘类型、LicenseType、Tag及附加一块100GB的数据磁盘）

备注：以下脚本中没有定义添加数据磁盘的部分，100GB的数据磁盘建议在原始机器中进行附加，再对原始机器进行捕获，然后使用Image创建的虚拟机就会自带100GB的数据磁盘

#设置初始变量

$cred = Get-Credential

$rgName = "hlmrge2"

$location = "chinaeast2"

$imageName = "hlmwin10e2t1-image-202107060954"

$storageType = 'Premium_LRS'

$myVnet = "hlmrge2-vnet"

$nsgName = "hlmwin10e2t1-nsg"

$vmSize = "Standard_D2s_v3"

$licenseType = "Windows_Client"

$diagStorageName = "hlmdiastoe2"

$vmName = "hlmwin10e2t7";

#读取镜像文件、虚拟网络、网络安全组信息

$image = Get-AzImage -ImageName $imageName -ResourceGroupName $rgName

$vnet = Get-AzVirtualNetwork -Name $myVnet -ResourceGroupName $rgName

$nsg = Get-AzNetworkSecurityGroup -ResourceGroupName $rgName -Name $nsgName

#设置虚拟机名称变量

$nicName = "$vmName-Nic" ;

$dataDiskName = "$vmName-DataDisk01" ;

#$ipName = "$vmName-Pip";

#建立虚拟网卡

$nic = New-AzNetworkInterface -Name $nicName -ResourceGroupName $rgName -Location $location -SubnetId $vnet.Subnets[0].Id -NetworkSecurityGroupId $nsg.Id ;

#建立VM配置文件（操作系统为 Windwos 系统磁盘为128GB）

$vm = New-AzVMConfig -VMName $vmName -VMSize $vmSize -LicenseType $licenseType -Tags @{"Daily Snapshot"="True"} ;

$vm = Set-AzVMSourceImage -VM $vm -Id $image.Id ;

$vm = Set-AzVMOSDisk -VM $vm -StorageAccountType $storageType -DiskSizeInGB 128 -CreateOption FromImage -Caching ReadWrite ;

$vm = Set-AzVMOperatingSystem -VM $vm -Windows -ComputerName $vmName -Credential $cred -ProvisionVMAgent -EnableAutoUpdate ;

$vm = Set-AzVMBootDiagnostic -VM $vm -Enable -ResourceGroupName $rgName -StorageAccountName $diagStorageName

$vm = Add-AzVMNetworkInterface -VM $vm -Id $nic.Id ;

#建立VM

New-AzVM -VM $vm -ResourceGroupName $rgName -Location $location;

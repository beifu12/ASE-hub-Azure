//使用当前的时间点减去最新的可用恢复点的时间

AzureDiagnostics

| where ResourceProvider == "MICROSOFT.RECOVERYSERVICES" and Category == "AzureSiteRecoveryRecoveryPoints"

| extend timeSinceLastRecoveryPoint = TimeGenerated - lastRecoveryPoint_t

| project TimeGenerated, timeSinceLastRecoveryPoint,CorrelationId

|join  (

| where Category == "AzureSiteRecoveryReplicatedItems"

|project name_s,CorrelationId

) on CorrelationId

| where timeSinceLastRecoveryPoint > 24h

| summarize arg_max(TimeGenerated, *) by CorrelationId

| project TimeGenerated, timeSinceLastRecoveryPoint,name_s

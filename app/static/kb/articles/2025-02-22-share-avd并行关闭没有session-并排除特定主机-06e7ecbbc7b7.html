问题现象 / 场景

背景信息

  • # Connect to Azure China Cloud and select the subscription
  • Connect-AzAccount -Environment AzureChinaCloud
  • Select-AzSubscription -SubscriptionId '97a20f90-b022-457f-95cd-72eeda*****'
  • # Define host pool and resource group
  • $HostPoolName = "autopoweroff"
  • $ResourceGroupName = "wvd"
  • # List of VMs to exclude from power-off
  • $notoffvms = @('autopoweroff/auto-0', 'autopoweroff/auto-1')
  • # Retrieve all session hosts in the host pool
  • $SessionHosts = Get-AzWvdSessionHost -HostPoolName $HostPoolName -ResourceGroupName $ResourceGroupName
  • # Process each session host
  • $SessionHosts | ForEach-Object -Parallel {
  • $SessionHost = $_
  • # Skip if the session host is in the exclusion list
  • if ($SessionHost.Name -in $notoffvms) {
  • Write-Output "Skipping excluded VM: $($SessionHost.Name)"
  • continue
  • }
  • # If the session count is 0, stop the VM
  • if ($SessionHost.Session -eq 0) {
  • Stop-AzVM -Id $SessionHost.ResourceId -Force
  • Write-Output "Host $($SessionHost.ResourceId) has no sessions and was stopped successfully."
  • elseif ($SessionHost.Session -gt 0) {
  • # Retrieve user sessions for the session host
  • $UserSessions = Get-AzWvdUserSession -HostPoolName $HostPoolName -ResourceGroupName $ResourceGroupName -SessionHostName $SessionHost.Name.Split("/")[1]
  • $hasActiveSession = $UserSessions.SessionState -contains "Active"
  • if ($hasActiveSession) {
  • Write-Output "Host $($SessionHost.ResourceId) has active sessions and will not be stopped."
  • } else {
  • Write-Output "Host $($SessionHost.ResourceId) has no active sessions and was stopped successfully."
  • } -ThrottleLimit 10 # Adjust throttle limit based on your environment

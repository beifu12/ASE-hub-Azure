添加一个runbook脚本

将如下命令中的参数更改为您环境中的信息后，粘贴到新弹出的窗口

#本脚本每周有一天创建完整快照，其他时间创建增量快照，保留30天

#操作的订阅ID,请根据实际修改

$SubscriptionID = "xxx";

#自动化中的凭据名称,请根据实际修改

$CredName = "xxx"

$cred = Get-Automationpscredential -Name $CredName;

Connect-AzAccount -Credential $cred -EnvironmentName AzureChinaCloud;

Select-AzSubscription -SubscriptionID $SubscriptionID;

#以下参数请按实际情况修改

#托管磁盘所在的资源组

$resourceGroupName = "xxx";

#托管磁盘的名称

$managedDiskName = "xxx";

#定义每周几创建完整快照，如Sunday

$fullSnapshotDay = "xxx"

#定义保留天数，如30

$Days = "x";

$snapshotName = "{0}_Snapshot_{1}" -f $managedDiskName, (Get-Date).ToString("yyyyMMddHHmmss");

$disk = Get-AzDisk -ResourceGroupName $resourceGroupName -DiskName $managedDiskName;

if ($disk -eq $null)

{

Write-Host "Disk not found";

return;

}

if((Get-Date).DayOfWeek.ToString() -eq $fullSnapshotDay)

Write-Host ("Creating full Snapshot for {0}, snapshot name: {1}" -f $managedDiskName, $snapshotName) -ForegroundColor Yellow;

$snapshotConfig = New-AzSnapshotConfig -Location $disk.Location -SourceUri $disk.Id -CreateOption Copy;

New-AzSnapshot -ResourceGroupName $resourceGroupName -SnapshotName $snapshotName -Snapshot $snapshotConfig

Write-Host "Full Snapshot Finished" -ForegroundColor Green;

else

Write-Host ("Creating Incremental Snapshot for {0}, snapshot name: {1}" -f $managedDiskName, $snapshotName) -ForegroundColor Yellow;

$snapshotConfig = New-AzSnapshotConfig -Location $disk.Location -SourceUri $disk.Id -CreateOption Copy -Incremental;

Write-Host "Incremental Snapshot Finished" -ForegroundColor Green;

$snapshots = Get-AzSnapshot -ResourceGroupName $resourceGroupName;

$filteredSnapshots = @($snapshots | where {$_.CreationData.SourceResourceId -eq $disk.Id});

Write-Host ("Snapshot found for disk {0}:" -f $ManagedDiskName);

$filteredSnapshots | select @{Name="TimeCreated"; Expression={$_.TimeCreated.ToLocalTime()}},Name,Location;

$timespan = New-TimeSpan -Days $Days

#Get current date

$current = get-date

#Declare retention time

$retention = $current-$timespan

#Delete snapshot before retention time

foreach ($filteredSnapshot in $filteredSnapshots)

if ($filteredSnapshot.TimeCreated -le $retention)

Remove-AzSnapshot -ResourceGroupName $resourceGroupName -SnapshotName $filteredSnapshot.Name -Force;

Write-Host ("Removesnapshot {0}:" -f $filteredSnapshot.Name)

点击“测试窗格”

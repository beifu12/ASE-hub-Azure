AKS RBAC设置如下

1、获取AKS信息

#获取当前连接的K8S集群的集群名称

kubectl config get-contexts | grep "*" | awk '{print $2}'

clustername=kubectl config get-contexts | grep "*" | awk '{print $2}'

#获取当前连接的K8S集群的API地址

kubectl cluster-info | grep "Kubernetes control plane" | awk '{print $NF}'

apiserver=kubectl cluster-info | grep "Kubernetes control plane" | awk '{print $NF}'

2、创建K8S RBAC

#特别说明RBAC中包含了EXEC POD的权限，就是可以通过exec命今进入到pods中查看日志

echo '''

apiVersion: v1

kind: ServiceAccount

metadata:

name: readonly

namespace: kube-system

---

apiVersion: rbac.authorization.k8s.io/v1

kind: ClusterRole

name: custom-cluster-readonly

rules:

  • apiGroups: ["*"]

resources: ["*"]

verbs: ["get","list","watch"]

resources: ["pods/exec"]

verbs: ["get","list","wateh","create"]

kind: ClusterRoleBinding

name: custom-clusterrolebingding-readonly

subjects:

  • kind: ServiceAccount

roleRef:

apiGroup: rbac.authorization.k8s.io

'''> RBAC.yaml

kubectl apply -f RBAC.yaml

#检查yaml http://nodeca.github.io/js-yaml/

3、获取Token

kind: Secret

annotations:

kubernetes.io/service-account.name: "readonly"

type: kubernetes.io/service-account-token

''' > token.yaml

kubectl apply -f token.yaml

#查看token创建是否成功

kubectl describe serviceaccount -n kube-system readonly

4、生成kubeconfig 文件

#打印token信息，这个等会产生kubeconfig文件的时候需要用到

kubectl get secret -n kube-system readonly -o "jsonpath={.data.token}" | base64 -d

token=kubectl get secret -n kube-system readonly -o "jsonpath={.data.token}" | base64 -d

#生成ca.crt文件

kubectl get secret -n kube-system readonly -o "jsonpath={.data['ca\.crt']}" | base64 -d > ca.crt

#这里的clustername和server地址就是文章一开始获取的clustername和api地址

Or if tls not needed, replace --certificate-authority and --embed-certs with

#注意配置文件位置如果不能正确指定在windows 中可以使用./.kube/kudev-readonly-config，测试时为了方便建议使用新的config 文件，不要修改默认的config

kubectl config set-cluster $clustername \

--embed-certs=true \

--certificate-authority=ca.crt \

--server=$apiserver \

--kubeconfig=~/.kube/kudev-readonly-config

#--token对应前面打印的token信息，可以使用$token 变量

kubectl config set-credentials readonly --token=$token --kubeconfig=~/.kube/kudev-readonly-config

#这里的clustername就是文章一开始获取的clustername

kubectl config set-context readonly --cluster=$clustername --user=readonly --kubeconfig=~/.kube/kudev-readonly-config

#将dev-readonly-config文件设置为使用readonly

kubectl config use-context readonly --kubeconfig=~/.kube/kudev-readonly-config

5、测试访问

#测试

kubectl --kubeconfig=~/.kube/kudev-readonly-config get pods

kubectl --kubeconfig=~/.kube/kudev-readonly-config delete pods web-0

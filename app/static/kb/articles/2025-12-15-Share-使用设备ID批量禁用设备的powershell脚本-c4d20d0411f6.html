参数

$InputCsvPath = "E:\deviceid.csv"       #输入表格位置

$OutputCsvPath = "E:\disabledeviceid.csv"    # 输出表格位置

$DeviceIdColumn = "DeviceId"  # 设备ID列名

从CSV读取设备ID

$csvDeviceIds = Import-Csv -Path $InputCsvPath | Select-Object -ExpandProperty $DeviceIdColumn

获取所有Entra ID设备

$allDevices = Get-MgDevice -All

找出不在CSV中且未禁用的设备

$devicesToDisable = $allDevices | Where-Object {

$_.DeviceId -notin $csvDeviceIds -and $_.AccountEnabled -ne $false

}

$devicesToDisable

Write-Host "找到 $($devicesToDisable.Count) 个需要禁用的设备"

禁用设备并记录

$disabledDevices = @()

foreach ($device in $devicesToDisable) {

Update-MgDevice -DeviceId $device.Id -BodyParameter @{accountEnabled = $false}

$disabledDevices += [PSCustomObject]@{

DeviceId = $device.DeviceId

DisplayName = $device.DisplayName

OperatingSystem = $device.OperatingSystem

DisabledDate = Get-Date -Format "yyyy-MM-dd HH:mm:ss"

Write-Host "已禁用: $($device.DisplayName) ($($device.DeviceId))"

======================================================================================

安装好Azcopy工具，默认的安装路径是“C:\'Program Files (x86)'\'Microsoft SDKs'\Azure\AzCopy\AzCopy.exe”请根据实际环境路径修改下面命令中对应的位置

安装Azure PowerShell工具以管理员身份运行，登陆订阅

修改下面命令标记黄色的参数，运行

==================================================

#复制的源存储账号名称

$SourceStorageAccountName = "xxx"

#复制的目标存储账号名称

$DestStorageAccountName = "xxxx"

#源存储账号访问密钥

$SourceStorageKey = "xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx=="

#目标存储账号访问密钥

$DestStorageKey = " xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx=="

#源存储账号变量环境

$StorageContext = New-AzureStorageContext -StorageAccountName $SourceStorageAccountName -StorageAccountKey $SourceStorageKey -Endpoint "core.chinacloudapi.cn"

#获取源存储账号中的容器

$containers = Get-AzureStorageContainer -Context $StorageContext

#循环执行每一个容器（上一步get到的）内容的复制工作

foreach ($c in $containers) {

"Transfer container " + $c.Name

$cmd = "C:\'Program Files (x86)'\'Microsoft SDKs'\Azure\AzCopy\AzCopy.exe /Source:" + $c.CloudBlobContainer.Uri.AbsoluteUri + " /Dest:https://" + $DestStorageAccountName + ".blob.core.chinacloudapi.cn/" + $c.Name + " /SourceKey:" + $SourceStorageKey + " /DestKey:" + $DestStorageKey + " /S"

Invoke-Expression $cmd

}

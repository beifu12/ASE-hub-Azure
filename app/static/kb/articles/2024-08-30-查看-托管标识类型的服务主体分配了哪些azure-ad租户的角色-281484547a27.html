  • 我这里的服务主体是托管标识类型的服务主体。
  • # 连接到 Azure AD
  • Connect-AzureAD -AzureEnvironmentName azurechinacloud
  • # 获取所有 Azure AD 目录角色
  • $AADRoles = Get-AzureADDirectoryRole
  • $Spname = "omsagent-k8s-3"
  • # 遍历每个角色
  • foreach ($AADRole in $AADRoles) {
  • # 获取当前角色的所有成员
  • $AADRoleMember = Get-AzureADDirectoryRoleMember -ObjectId $AADRole.ObjectId
  • # 只保留分配了特定服务主体的角色
  • $servicePrincipalsInRole = $AADRoleMember | Where-Object { $_.ObjectType -eq "ServicePrincipal" -and $_.DisplayName -eq "$Spname" }
  • # 如果存在服务主体，将角色信息和服务主体信息输出
  • if ($servicePrincipalsInRole) {
  • Write-Output "角色名称: $($AADRole.DisplayName)"
  • Write-Output "角色 ObjectId: $($AADRole.ObjectId)"
  • Write-Output "以下是分配到该角色的服务主体:"
  • foreach ($servicePrincipal in $servicePrincipalsInRole) {
  • Write-Output "服务主体名称: $($servicePrincipal.DisplayName) - $($servicePrincipal.UserPrincipalName)"
  • }
  • Write-Output "-------------------------------"

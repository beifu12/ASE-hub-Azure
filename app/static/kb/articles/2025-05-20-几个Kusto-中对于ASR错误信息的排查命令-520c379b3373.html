// ASR errors

cluster('asradxclusmc.chinanorth2.kusto.chinacloudapi.cn').database('ASRKustoDB').

| where resourceId startswith "/SUBSCRIPTIONS/5d24c88a-e104-419a-a9cd-b2563ded04d8" // Insert your subscription id here

| where PreciseTimeStamp > datetime(2020-08-13 00:00:00.000)

| where PreciseTimeStamp < datetime(2020-09-14 00:00:00.000)

| where properties contains "ServerHealth"

| project PreciseTimeStamp, p = parsejson(properties)

| project PreciseTimeStamp, p.affectedResourceName, p.timeOfOccurence, p.description, p.healthErrors

| mv-expand errors = p_healthErrors

//| where p_affectedResourceName contains "GPM-ZIJIN-MYSQL"

| project PreciseTimeStamp, p_affectedResourceName, p_timeOfOccurence, p_description, errors.creationTime, errors.errorCode, errors.errorMessage

| order by PreciseTimeStamp desc

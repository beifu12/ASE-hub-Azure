$user = ""

$password = ConvertTo-SecureString -String "" -AsPlainText -Force

$credential = New-Object -TypeName "System.Management.Automation.PSCredential" -ArgumentList $user,$password

Connect-AzureAD -AzureEnvironmentName AzureChinaCloud -Credential $credential

Connect-MsolService -AzureEnvironment AzureChinaCloud -Credential $credential

Connect-MgGraph -ClientId -TenantId -CertificateThumbprint -Environment China

$users = Get-MgUser -All -Property DisplayName,lastPasswordChangeDateTime,PasswordPolicies | select DisplayName,lastPasswordChangeDateTime

foreach ($user in $users){

$policyset=Get-MsolPasswordPolicy -DomainName

$poly=$policyset.ValidityPeriod

$passwordLastSet = $user.LastPasswordChangeDateTime.Date

$expiryDate = $passwordLastSet.AddDays($poly)

$daysLeft = ($expiryDate - (Get-Date)).Days

Write-Output "$($user.DisplayName): Password expires in $daysLeft days."

}

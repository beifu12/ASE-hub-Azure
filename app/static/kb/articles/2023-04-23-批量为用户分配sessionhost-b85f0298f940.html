脚本参考：

for($i=0;$i -le 1;$i++)

{

$user=Get-AzADGroupMember -GroupObjectId f7e5aa79-2a93-4754-ac38-ed00a8611826 |  Select DisplayName,UserPrincipalName

$upn=Get-AzADUser -DisplayName $user.DisplayName[$i]

$sessionhost=Get-AzWvdSessionHost -HostPoolName avd317  -ResourceGroupName avd317

$sessionhost2=$sessionhost.Name.Split("/")[2*($i)+1]

Update-AzWvdSessionHost -HostPoolName avd317  -Name $sessionhost2  -ResourceGroupName avd317 -AssignedUser $upn.UserPrincipalName

Write-Output "$i"

}

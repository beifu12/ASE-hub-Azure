Case num：20220113143762

terraform版本有较新版本，之前0.几的版本在部署时会提示版本太老。当前最新版为1.1.3，provider版本时2.91.0.

部署terraform wget https://releases.hashicorp.com/terraform/1.1.3/terraform_1.1.3_linux_amd64.zip

Unzip terraform_1.1.3_linux_amd64.zip

解压后将terraform脚本放在/usr/bin下即可。

新版本在指定terraform版本的语法上发生改变，以下为使用共享镜像库中的镜像创建Linux虚拟机的脚本。

terraform {

required_providers {

azurerm = {

source = "hashicorp/azurerm"

version = "2.90.0"

}

provider "azurerm" {

features {}

subscription_id = "a07125f6-e353-418a-ab1c-d29a656ae44d"

client_id = "27bbc68a-15ef-4173-9cc7-eaa634235822" //appid

client_secret = "2ZPRexq8KN7OB-dZ1R3LK.3mkX1D.UB2h." //paaaword

tenant_id = "b388b808-0ec9-4a09-a414-a7cbbd8b7e9b" //tenant id

environment = "china"

data "azurerm_shared_image" "example" {

name = "centosimage"

gallery_name = "image"

resource_group_name = "cdn"

resource "azurerm_resource_group" "example" {

name = "testrg"

location = "China East2"

resource "azurerm_virtual_network" "example" {

name = "example-network"

address_space = ["10.0.0.0/16"]

location = azurerm_resource_group.example.location

resource_group_name = azurerm_resource_group.example.name

resource "azurerm_subnet" "example" {

name = "internal"

resource_group_name = azurerm_resource_group.example.name

virtual_network_name = azurerm_virtual_network.example.name

address_prefixes = ["10.0.2.0/24"]

resource "azurerm_network_interface" "example" {

name = "example-nic"

ip_configuration {

name = "internal"

subnet_id = azurerm_subnet.example.id

private_ip_address_allocation = "Dynamic"

resource "azurerm_linux_virtual_machine" "example" {

name = "example-machine"

size = "Standard_F2"

disable_password_authentication = "false"

admin_username = "wzq"

admin_password = "!QAZ2wsx3edc"

network_interface_ids = [

azurerm_network_interface.example.id,

]

os_disk {

caching = "ReadWrite"

storage_account_type = "Standard_LRS"

source_image_id = data.azurerm_shared_image.example.id

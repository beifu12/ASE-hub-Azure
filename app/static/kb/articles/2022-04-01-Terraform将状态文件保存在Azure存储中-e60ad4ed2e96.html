中国区目前没有相关文档，可参考global文档，但是连接中国区时有以下注意的地方。除了在provider中指定连接中国区外，在backend中也要指定中国区，否则在terraform init时就是报如下错误：

调整后脚本如下：

terraform {

required_providers {

azurerm = {

source = "hashicorp/azurerm"

version = "2.90.0"

}

backend "azurerm" {

resource_group_name = "cdn"

storage_account_name = "testac"

container_name = "tfstate"

key = "myterraform.tfstate"

environment = "china"

provider "azurerm" {

features {}

subscription_id = "a07125f6-e353-418a-ab1c-d29a656ae44d"

client_id = "27bbc68a-15ef-4173-9cc7-eaa634235822"

client_secret = "3.os3jpjrmZp6wlT3H4~.JiG-TRuW9Y~Ki"

tenant_id = "b388b808-0ec9-4a09-a414-a7cbbd8b7e9b"

resource "azurerm_resource_group" "state-demo-secure" {

name = "state-demo"

location = "chinanorth2"

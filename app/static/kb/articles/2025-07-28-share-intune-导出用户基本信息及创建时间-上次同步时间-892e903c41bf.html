Install-Module Microsoft.Graph

Import-Module Microsoft.Graph

$clientId = "c3ad0400-8b2b-45d0-9915-d87383****e"

$clientSecret = "qo.8ahG8_Hy2zq3-GoV_6-51*****I"

$tenantId = "b388b808-0ec9-4a09-a414-a7c*******"

$securePassword = ConvertTo-SecureString $clientSecret -AsPlainText -Force

$credential = New-Object System.Management.Automation.PSCredential($clientId, $securePassword)

Connect-MgGraph -ClientSecretCredential $credential -TenantId $tenantId -Environment China -NoWelcome -Verbose -ErrorAction Stop

获取用户列表

$users = Get-MgUser -All -Property "id,displayName,userPrincipalName,createdDateTime,lastPasswordChangeDateTime"

导出到CSV

$users | Select-Object id,displayName,userPrincipalName,createdDateTime,lastPasswordChangeDateTime | Export-Csv -Path "C:\UsersList.csv" -NoTypeInformation -Encoding UTF8

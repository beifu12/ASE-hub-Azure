Connect-AzAccount -Environment AzureChinaCloud

#—— 输入参数 Firewall Policy Name, Firewall Policy Resource Group ——

$fpname = "Firewall Policy Name"

$fprg   = "Firewall Policy Resource Group"

$fprcgNATname = "DefaultNetworkRuleCollectionGroup"

$fprcgAPPname = "DefaultApplicationRuleCollectionGroup"

#—— 获取防火墙策略对象 ——

$fp = Get-AzFirewallPolicy -Name $fpname -ResourceGroupName $fprg

#—— 获取该策略下的Nat Rule Collection Group ——

$rcgNAT = Get-AzFirewallPolicyRuleCollectionGroup -Name $fprcgNATname -AzureFirewallPolicy $fp

$rcgApp = Get-AzFirewallPolicyRuleCollectionGroup -Name $fprcgAPPname -AzureFirewallPolicy $fp

#—— 确保输出目录存在 ——

$outputRoot = Join-Path -Path (Get-Location) -ChildPath $fpname

if (-not (Test-Path $outputRoot)) {

New-Item -ItemType Directory -Path $outputRoot | Out-Null

}

$returnObjNAT  = @()

$returnObjAPP  = @()

#—— 遍历每个 Rule Collection Group 并导出 ——

foreach ($rulecol in $rcgNAT.Properties.RuleCollection) {

foreach ($rule in $rulecol.Rules) {

$properties = [ordered]@{

RulePriority        = $rulecol.Priority;

RuleCollectionName  = $rulecol.Name;

RuleType            = $rulecol.RuleCollectionType;

ActionType          = $rulecol.Action.Type;

RuleName            = $rule.Name;

SourceAddresses     = $rule.SourceAddresses     -join ", ";

SourceIPGroups      = $rule.SourceIPGroups      -join ", ";

DestinationPorts    = $rule.DestinationPorts    -join ", ";

Protocols           = $rule.Protocols           -join ", ";

DestinationAddresses= $rule.DestinationAddresses-join ", ";

DestinationIPGroups = $rule.DestinationIPGroups -join ", ";

DestinationFQDNs    = $rule.DestinationFQDNs    -join ", ";

$obj = New-Object psobject -Property $properties

$returnObjNAT += $obj

构造输出文件路径，并导出 CSV

$csvPath = Join-Path -Path $outputRoot -ChildPath "$($rcgNAT.Name).csv"

$returnObjNAT | Export-Csv -Path $csvPath -NoTypeInformation -Encoding UTF8

Write-Host "已导出规则集合组 '$($rcgNAT.Name)' 到文件：$csvPath"

foreach ($rulecol in $rcgApp.Properties.RuleCollection) {

Protocols = ($rule.Protocols |ForEach-Object { "$($_.ProtocolType):$($_.Port)" }) -join ',';

TargetFqdns         = $rule.TargetFqdns         -join ", ";

TargetUrls          = $rule.TargetUrls          -join ", ";

FqdnTags            = $rule.FqdnTags            -join ", ";

WebCategories       = $rule.WebCategories       -join ", ";

TerminateTLS        = $rule.TerminateTLS        -join ", ";

HttpHeadersToInsert = $rule.HttpHeadersToInsert -join ", ";

$returnObjAPP += $obj

$csvPath = Join-Path -Path $outputRoot -ChildPath "$($rcgApp.Name).csv"

$returnObjAPP | Export-Csv -Path $csvPath -NoTypeInformation -Encoding UTF8

Write-Host "已导出规则集合组 '$($rcgApp.Name)' 到文件：$csvPath"

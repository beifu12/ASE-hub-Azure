创建一个rolebinding对这个sa进行授权：

vi ingress-appgw-sa-rolebinding.yaml

apiVersion: rbac.authorization.k8s.io/v1

kind: ClusterRoleBinding

metadata:

name: ingress-appgw-clusterrolebinding

  • kind: ServiceAccount

name: ingress-appgw-sa-ingress-azure # name of your service account

namespace: default # this is the namespace your service account is in

roleRef: # referring to your ClusterRole

kind: ClusterRole

name: cluster-admin

apiGroup: rbac.authorization.k8s.io

k get clusterrolebinding

NAME ROLE AGE

ingress-appgw-clusterrolebinding

#定义添加的端口，这里定义100个NAT规则，

$ports=30001..30100 #端口范围

#创建一个规则集并添加一个规则，如果有规则集的话，则不用创建了规则集了

$rule1 = New-AzureRMFirewallNatRule -Name R1 -Protocol "UDP","TCP" -SourceAddress "*" -DestinationAddress "40.73.110.69" -DestinationPort "30000" -TranslatedAddress "10.100.128.4" -TranslatedPort "30000"

$ruleCollection = New-AzureRMFirewallNatRuleCollection -Name "MyNatRuleCollection" -Priority 100 -Rule $rule1

$fw=Get-AzureRmFirewall -Name fw -ResourceGroupName fuzlrg

#向规则集中添加规则

foreach($port in $ports)

{

$name="dnat"+$port

#添加规则

$rule = New-AzureRMFirewallNatRule -Name $name  -Protocol "TCP","UDP" -SourceAddress "*" -DestinationAddress "40.73.110.69" -DestinationPort $port -TranslatedAddress "10.100.128.4" -TranslatedPort $port

$ruleCollection.AddRule($rule)

}

#向FW添加规则集并保存配置

$fw.AddNatRuleCollection($ruleCollection)

Set-AzureRmFirewall -AzureFirewall $fw
